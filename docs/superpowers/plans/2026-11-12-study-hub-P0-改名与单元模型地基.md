# 学习中枢 App（study-hub）—— P0 改名与单元模型地基实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 把 `projects/apps/algo-plan/` 整体改名为 `projects/apps/study-hub/`，并铺好「学习单元（Unit）」的地基——三条顶层轨的定义、Unit 模型与校验、三个 loader（A 算法题 / B grokking 卡 / C 新单元文件）、Markdown→HTML、单元索引缓存、SQLite 的 5 张新表与 7 条加列、`GET /api/units` 两条路由，以及 4 张数据库卡的断言脚本。做完这期，app 仍然是一个能跑、能判、能复习的算法学习 app，只是从内部长出了「三轨合一」的骨架。

**Architecture:** 内容侧三个来源（`learning/bank/**` 的算法题、`learning/scaffold/grokking/**` 的 29 张 Python 卡、`learning/units/**` 的新单元文件）被三个 loader 归一成同一个 `Unit` 对象，`core/unit/index.js` 按 **C → A → B** 的顺序去重合并；`core/track/tracks.js` 是「轨」的唯一定义处。调度态（作答流水、复习卡、会话、队列、掌握度）全部落 SQLite，新表与新列**只做加法**，既有 4 张表一个列名都不改。`core/unit/markdown.js` 是自写的极简 Markdown→HTML（零依赖铁律），供 `GET /api/units/:id` 的 `statement_html` 用。

**Tech Stack:** Node.js ≥ 24（内置 `node:sqlite` / `node:http` / `node:child_process` / `node:test` / `node:fs` / `node:path`）、原生 JS + CSS、**零 npm 依赖**、**零构建**、零前端框架、零第三方图表库。

```bash
"use strict"
/**
 * paths.js —— learning 仓根路径解析
 *
 * 不硬编码 learning 仓位置。只认环境变量 STUDY_HUB_LEARNING_ROOT，
 * 取不到就降级为「纯本地模式」：只有 SQLite，不与 learning 仓同步。
 *
 * 环境变量改名（G26 / G40）：旧的 ALGO_PLAN_LEARNING_ROOT 与
 * ALGO_PLAN_LOCAL_BANK 已在 P0 一次性换成 STUDY_HUB_*，代码里不留兼容分支。
 * 启动时「旧名字还在」的提示放在 server/main.js 的 checkLegacyEnvNames()。

const fs = require("node:fs")
const path = require("node:path")

/**
 * 解析 learning 仓根目录。
 * @param {object} env 进程环境变量对象
 * @param {string} [override] 显式指定路径（优先于环境变量，测试用）
 * @returns {{root: string|null, ok: boolean, reason: string}}
 */
function resolveLearningRoot(env, override) {
  const raw =
    override !== undefined && override !== null
      ? override
      : (env || process.env).STUDY_HUB_LEARNING_ROOT

  if (raw === undefined || raw === null || String(raw).trim() === "") {
    return {
      root: null,
      ok: false,
      reason: "未设置 STUDY_HUB_LEARNING_ROOT，进入纯本地模式（不与 learning 仓同步）",
    }
  }

  const abs = path.resolve(String(raw).trim())
  if (!fs.existsSync(abs)) {
    return { root: null, ok: false, reason: "STUDY_HUB_LEARNING_ROOT 指向的目录不存在：" + abs }
  }
  if (!fs.statSync(abs).isDirectory()) {
    return { root: null, ok: false, reason: "STUDY_HUB_LEARNING_ROOT 不是目录：" + abs }
  }
  return { root: abs, ok: true, reason: "" }
}

/**
 * 题库目录：learning 仓下的 bank/（loader A 的源之一）。
 * @param {string|null} learningRoot
 * @returns {string|null}
 */
function bankDir(learningRoot) {
  if (!learningRoot) return null
  return path.join(learningRoot, "bank")
}

/**
 * 新单元目录：learning 仓下的 units/（loader C 的唯一落点，P4 起由供给流水线写入）。
 * @param {string|null} learningRoot
 * @returns {string|null}
 */
function unitsDir(learningRoot) {
  if (!learningRoot) return null
  return path.join(learningRoot, "units")
}

/**
 * grokking 卡目录：learning 仓下的 scaffold/grokking/（loader B 的源）。
 * @param {string|null} learningRoot
 * @returns {string|null}
 */
function grokkingDir(learningRoot) {
  if (!learningRoot) return null
  return path.join(learningRoot, "scaffold", "grokking")
}

/**
 * 工程仓内置题库根：apps/study-hub/bank/。
 *
 * 为什么要有第二套题库：
 *   learning/bank 放人审内容（题解、译文），改动要进 git；
 *   而 Codeforces 这类「机器抓来的原题」不适合塞进人审仓——
 *   51 个文件一次性涌入，人审仓的 diff 全是噪音。
 *   所以机器题单独放工程仓，跟代码一起走版本控制，两边都能取到。
 *
 * 环境变量 STUDY_HUB_LOCAL_BANK 可以把它指到别处。
 * @param {object} env 进程环境变量对象
 * @param {string} [override] 显式指定路径（优先于环境变量，测试用）
```

```bash
 *
 * 字段表见契约 §1.2。本文件是**唯一**定义 Unit 形状的地方。
 */

// 轨道 → id 前缀的映射只有 core/track/tracks.js 一份（G41），这里引用它，不另抄一份。
const { TRACKS, trackOf } = require("../track/tracks.js")

/** 单元种类：problem = 要写答案；card = 读讲解 + 跑 demo + 过断言。 */
const KINDS = ["problem", "card"]

/** 内容可信度。 */
const STATUSES = ["todo", "draft", "verified", "rejected"]

/** 学习进度（旧 frontmatter 的 status 语义）。 */
const PROGRESSES = ["todo", "in-progress", "done"]

/** 难度。 */
const DIFFICULTIES = ["easy", "medium", "hard"]

/** 入口种类。 */
const GRADED = ["user", "script"]

/** 内容来源。 */
const SOURCES = [
  "leetcode",
  "nowcoder",
  "self",
  "grokking",
  "scaffold",
  "generated",
  "codeforces",
  "unknown",
]

/**
 * 单元 id 的形状：<trackCode>:<stem>。
 * 前缀集由 tracks.js 派生，不另抄一份——抄一份就会漂移（G41）。
 */
const ID_PATTERN = new RegExp(
  "^(" + TRACKS.map(function (t) { return t.idPrefix }).join("|") + "):[a-z0-9][a-z0-9._-]*$"
)

/**
 * 归一化一个字符串数组：去空、去首尾空白、去重、保序。
 * @param {string[]|string|null|undefined} list
 * @returns {string[]}
 */
function normalizeList(list) {
  const input = Array.isArray(list) ? list : list === undefined || list === null ? [] : [list]
  const out = []
  for (const item of input) {
    if (item === undefined || item === null) continue
    const text = String(item).trim()
    if (text === "") continue
    if (out.indexOf(text) !== -1) continue
    out.push(text)
  }
  return out
}

/**
 * 把任意来源的 id 规范成 <trackCode>:<stem>。
 * 已经带前缀的原样返回；不带前缀的按 track 补前缀。
 *
 * 轨道 → id 前缀的映射只有 core/track/tracks.js 一份（G41），这里引用它，不另抄一份。
 * 未知轨道返回空串——**不许静默兜底成 algo 前缀**：那样会造出前缀与轨道不符、
 * 永远 join 不上 SQLite 调度态的单元（G24）。
 *
 * 注意：规范化只发生在**内存层**——存量内容文件里的 id 一个字都不改（G33）。
 *
 * @param {string} raw 原始 id
 * @param {string} track 轨 code（algorithm / database / ai）
 * @returns {string}
 */
function normalizeUnitId(raw, track) {
  const text = String(raw === undefined || raw === null ? "" : raw).trim()
  if (text === "") return ""
  // 轨道先查：未知轨道一律返回空串（G41）——连「已经带前缀的 raw」也不例外，
  // 否则会造出前缀与轨道对不上的 id。
  const t = trackOf(track)
  if (!t) return ""
  if (text.indexOf(":") !== -1) return text
  return t.idPrefix + ":" + text
}

/**
 * 把各种形态的真假值归一成布尔。
 * 只有 true 与字符串 "true" 算真——frontmatter 里的 `reviewed: false` 会被 mini-YAML
 * 解析成字符串 "false"，直接 Boolean() 会得到 true，所以必须显式处理。
 *
 * @param {*} value
 * @returns {boolean}
 */
function toBool(value) {
  if (value === true) return true
  if (typeof value === "string") return value.trim().toLowerCase() === "true"
  return false
}

/**
 * 校验一个单元对象。
 * @param {object} unit
 * @returns {string[]} 错误信息数组，空数组表示通过
 */
function validateUnit(unit) {
  const errors = []
  if (!unit || typeof unit !== "object") return ["单元不是对象"]
  if (!ID_PATTERN.test(String(unit.id || ""))) {
    errors.push("id 非法：" + String(unit.id || ""))
  }
  const t = trackOf(unit.track)
  if (!t) {
    errors.push("track 非法：" + String(unit.track || ""))
  } else {
    // G41：id 前缀必须与轨道一致。前缀对不上的单元永远 join 不上 SQLite 里的调度态，
    // 还不许落库——否则它会安静地待在那里，join 一次也 join 不上。
    const expected = t.idPrefix
    const prefix = String(unit.id || "").split(":")[0]
    if (prefix !== expected) {
      errors.push(
        "id 前缀与 track 不一致：" + String(unit.id || "") + "（" + unit.track + " 轨应是 " + expected + " 前缀）"
      )
    }
  }
  if (KINDS.indexOf(unit.kind) === -1) {
    errors.push("kind 非法：" + String(unit.kind || ""))
  }
  if (!String(unit.title || "").trim()) {
    errors.push("title 不能为空")
  }
  // 契约 §1.2：path 必填，空串视为缺失（G42）。没有 path 的单元下游找不到文件。
  if (!String(unit.path || "").trim()) {
    errors.push("path 不能为空")
  }
  if (!unit.entry || typeof unit.entry !== "object") {
    errors.push("entry 缺失")
  } else {
    if (GRADED.indexOf(unit.entry.graded) === -1) {
      errors.push("entry.graded 非法：" + String(unit.entry.graded || ""))
    }
    if (unit.entry.graded === "script" && !String(unit.entry.script || "").trim()) {
      errors.push("entry.graded 是 script 但 entry.script 为空")
    }
  }
  if (!Array.isArray(unit.asserts)) {
    errors.push("asserts 必须是数组")
  }
  if (!Array.isArray(unit.topics)) {
    errors.push("topics 必须是数组")
  }
  if (!Array.isArray(unit.subtracks)) {
    errors.push("subtracks 必须是数组")
  }
  if (DIFFICULTIES.indexOf(unit.difficulty) === -1) {
    errors.push("difficulty 非法：" + String(unit.difficulty || ""))
  }
  if (SOURCES.indexOf(unit.source) === -1) {
    errors.push("source 非法：" + String(unit.source || ""))
  }
  if (STATUSES.indexOf(unit.status) === -1) {
    errors.push("status 非法：" + String(unit.status || ""))
  }
  if (PROGRESSES.indexOf(unit.progress) === -1) {
    errors.push("progress 非法：" + String(unit.progress || ""))
  }
  if (typeof unit.reviewed !== "boolean") {
    errors.push("reviewed 必须是布尔值")
  }
  return errors
}
```

"use strict"
/**
 * markdown.js —— 极简 Markdown → HTML
 *
 * 零依赖铁律：不引 Markdown 库（契约 §8.1）。
 * 支持的语法就那几样：标题、段落、无序/有序列表、粗体、行内代码、代码块、表格、引用。
 * **未知语法原样输出为文本**——不报错、不丢内容。
 *
 * 输出是给单元学习页左栏用的**片段**，不含 <html>/<body> 外壳。
 */

/** 行内代码的占位哨兵：先抽出行内代码，转义完再放回去，免得代码里的 * 和 _ 被当成强调。 */
const SENTINEL = "\u0000"

/**
 * HTML 转义。
 * @param {*} text
 * @returns {string}
 */
function escapeHtml(text) {
  return String(text === undefined || text === null ? "" : text)
    .replace(/&/g, "&amp;")
    .replace(/</g, "&lt;")
    .replace(/>/g, "&gt;")
    .replace(/"/g, "&quot;")
    .replace(/'/g, "&#39;")
}

/**
 * 行内渲染：行内代码、粗体、斜体。
 * @param {string} text
 * @returns {string}
 */
function inline(text) {
  const codes = []
  let out = String(text === undefined || text === null ? "" : text).replace(/`([^`]+)`/g, function (_, code) {
    codes.push(code)
    return SENTINEL + (codes.length - 1) + SENTINEL
  })
  out = escapeHtml(out)
  out = out.replace(/\*\*([^*]+)\*\*/g, "<strong>$1</strong>")
  out = out.replace(/(^|[^*])\*([^*\n]+)\*/g, "$1<em>$2</em>")
  out = out.replace(/\u0000(\d+)\u0000/g, function (_, index) {
    return "<code>" + escapeHtml(codes[Number(index)]) + "</code>"
  })
  return out
}

/**
 * 判断一行是不是表格分隔行（|---|---| 或 --- | ---）。
 * @param {string} line
 * @returns {boolean}
 */
function isTableSeparator(line) {
  const text = String(line === undefined || line === null ? "" : line).trim()
  if (text.indexOf("-") === -1) return false
  return /^\|?\s*:?-+:?\s*(\|\s*:?-+:?\s*)*\|?$/.test(text)
}

/**
 * 判断当前位置是不是表格的表头行（当前行有 |，且下一行是分隔行）。
 * @param {string} line
 * @param {string} next
 * @returns {boolean}
 */
function isTableStart(line, next) {
  if (String(line === undefined || line === null ? "" : line).indexOf("|") === -1) return false
  return isTableSeparator(next)
}

/**
 * 切一行表格为单元格数组。单元格里写 `\|` 表示竖线本身。
 * @param {string} line
 * @returns {string[]}
 */
function splitTableRow(line) {
  let body = String(line === undefined || line === null ? "" : line).trim()
  if (body.charAt(0) === "|") body = body.slice(1)
  if (body.length > 0 && body.charAt(body.length - 1) === "|") body = body.slice(0, -1)
  const cells = []
  let buf = ""
  for (let i = 0; i < body.length; i++) {
    const ch = body.charAt(i)
    if (ch === "\\" && body.charAt(i + 1) === "|") {
      buf += "|"
      i += 1
      continue
    }
    if (ch === "|") {
      cells.push(buf.trim())
      buf = ""
      continue
    }
    buf += ch
  }
  cells.push(buf.trim())
  return cells
}

/**
 * 表格渲染。
 * @param {string[][]} rows 第一行是表头
 * @returns {string}
 */
function renderTable(rows) {
  if (!rows || rows.length === 0) return ""
  let out = "<table><thead><tr>"
  for (const cell of rows[0]) {
    out += "<th>" + inline(cell) + "</th>"
  }
  out += "</tr></thead><tbody>"
  for (const row of rows.slice(1)) {
    out += "<tr>"
    for (const cell of row) {
      out += "<td>" + inline(cell) + "</td>"
    }
    out += "</tr>"
  }
  out += "</tbody></table>"
  return out
}

/**
 * Markdown → HTML。
 * @param {string} markdown
 * @returns {string}
 */
function toHtml(markdown) {
  const lines = String(markdown === undefined || markdown === null ? "" : markdown).split(/\r?\n/)
  const out = []
  let i = 0

  while (i < lines.length) {
    const line = lines[i]

    // 代码块
    if (/^```/.test(line)) {
      const body = []
      i += 1
      while (i < lines.length && !/^```/.test(lines[i])) {
        body.push(lines[i])
        i += 1
      }
      i += 1 // 跳过收尾的 ```
      out.push("<pre><code>" + escapeHtml(body.join("\n")) + "</code></pre>")
      continue
    }

    // 标题
    const head = /^(#{1,6})\s+(.*)$/.exec(line)
    if (head) {
      const level = head[1].length
      out.push("<h" + level + ">" + inline(head[2]) + "</h" + level + ">")
      i += 1
      continue
    }

    // 表格
    if (isTableStart(line, lines[i + 1])) {
      const rows = [splitTableRow(line)]
      i += 2
      while (i < lines.length && lines[i].trim() !== "" && lines[i].indexOf("|") !== -1) {
        rows.push(splitTableRow(lines[i]))
        i += 1
      }
      out.push(renderTable(rows))
      continue
    }

    // 引用
    if (/^>\s?/.test(line)) {
      const body = []
      while (i < lines.length && /^>\s?/.test(lines[i])) {
        body.push(lines[i].replace(/^>\s?/, ""))
        i += 1
      }
      out.push("<blockquote>" + toHtml(body.join("\n")) + "</blockquote>")
      continue
    }

    // 列表
    if (/^\s*[-*]\s+/.test(line) || /^\s*\d+\.\s+/.test(line)) {
      const ordered = /^\s*\d+\.\s+/.test(line)
      const items = []
      while (i < lines.length) {
        const m = ordered
          ? /^\s*\d+\.\s+(.*)$/.exec(lines[i])
          : /^\s*[-*]\s+(.*)$/.exec(lines[i])
        if (!m) break
        items.push(m[1])
        i += 1
      }
      const tag = ordered ? "ol" : "ul"
      let html = "<" + tag + ">"
      for (const item of items) {
        html += "<li>" + inline(item) + "</li>"
      }
      html += "</" + tag + ">"
      out.push(html)
      continue
    }

    // 空行
    if (line.trim() === "") {
      i += 1
      continue
    }

    // 段落：连续的非特殊非空行合成一段
    const para = []
    while (
      i < lines.length &&
      lines[i].trim() !== "" &&
      !/^(#{1,6})\s/.test(lines[i]) &&
      !/^```/.test(lines[i]) &&
      !/^>\s?/.test(lines[i]) &&
      !/^\s*[-*]\s+/.test(lines[i]) &&
      !/^\s*\d+\.\s+/.test(lines[i]) &&
      !isTableStart(lines[i], lines[i + 1])
    ) {
      para.push(lines[i])
      i += 1
    }
    if (para.length > 0) {
      out.push("<p>" + inline(para.join("\n")).replace(/\n/g, "<br>") + "</p>")
    } else {
      i += 1
    }
  }

  return out.join("\n")
}

module.exports = {
  escapeHtml: escapeHtml,
  inline: inline,
  isTableSeparator: isTableSeparator,
  splitTableRow: splitTableRow,
  toHtml: toHtml,
}

```bash
"use strict"
/**
 * loader-a.js —— 算法题 md → Unit
 *
 * 读 learning/bank 下的 md 与工程仓 bank 下的 md，产出与 DB/AI 卡**完全同构**的 Unit。
 * 契约 §3.1 冻结了本文件的行为。
 *
 * 铁律：**只读**。存量题库文件一个字都不改（G32）——id 的规范化只发生在内存层（G33）。
 */

const { parseProblemMarkdown } = require("../bank/parse")
// Unit 的枚举只有 core/unit/unit.js 一份（G41）：这里引用，不另抄。
const { createUnit, PROGRESSES, DIFFICULTIES, SOURCES, normalizeUnitId } = require("./unit")

/**
 * 按内容判定单元状态（契约 §1.4 的 loader A 规则）。
 *
 * 刻意**不看** frontmatter 的 needs_fill —— 那个标志在存量里已经过时：
 * 51 道 CF 原题的 needs_fill 仍是 true，可题面与用例其实都齐。
 *
 * @param {object} problem
 * @returns {"todo"|"draft"|"verified"}
 */
function statusOf(problem) {
  const hasStatement = typeof problem.statement === "string" && problem.statement.trim() !== ""
  const hasCases = Array.isArray(problem.cases) && problem.cases.length > 0
  if (!hasStatement) return "todo"
  if (!hasCases) return "draft"
  return "verified"
}

/**
 * 来源归一化：不在枚举里 → unknown。
 * 来源表只有 core/unit/unit.js 的 SOURCES 一份（G41），这里引用，不另抄。
 * @param {*} raw
 * @returns {string}
 */
function normalizeSource(raw) {
  const value = String(raw === undefined || raw === null ? "" : raw).trim()
  return SOURCES.indexOf(value) !== -1 ? value : "unknown"
}

/**
 * 学习进度归一化：不在枚举里 → todo。
 * @param {*} raw
 * @returns {string}
 */
function progressOf(raw) {
  const value = String(raw === undefined || raw === null ? "" : raw).trim()
  return PROGRESSES.indexOf(value) !== -1 ? value : "todo"
}

/**
 * 难度归一化：不在枚举里 → medium。
 * @param {*} raw
 * @returns {string}
 */
function difficultyOf(raw) {
  const value = String(raw === undefined || raw === null ? "" : raw).trim()
  return DIFFICULTIES.indexOf(value) !== -1 ? value : "medium"
}

/**
 * 解析一份算法题 md，产出 Unit。
 *
 * @param {string} text 文件全文
 * @param {object} opts { track?: string, relPath: string }（track 缺省为 "algorithm"）
 * @returns {{unit: object|null, warnings: string[]}}
 */
function parseUnitMarkdown(text, opts) {
  const options = opts || {}
  const warnings = []

  const parsed = parseProblemMarkdown(text)
  ;(parsed.warnings || []).forEach(function (w) {
    warnings.push(w)
  })

  const problem = parsed.problem
  if (!problem || !problem.id) {
    warnings.push("算法题解析不出 id，跳过")
    return { unit: null, warnings: warnings }
  }

  // G42：Unit.path 必填。调用方必须给出文件相对内容根的路径（loader 不自己拼路径），
  // 缺了就只给一条警告，不产出单元——空的 path 会让下游找不到文件。
  if (!String(options.relPath || "").trim()) {
    warnings.push("算法题 " + problem.id + " 缺少 relPath，无法产出单元")
    return { unit: null, warnings: warnings }
  }

  // 轨道来自调用方（缺省算法轨）。轨道↔id 前缀的映射只有 core/track/tracks.js 一份（G41），
  // 所以 id 一律走 core/unit/unit.js 的 normalizeUnitId，loader 自己不拼前缀。
  const track = String(options.track || "algorithm").trim() || "algorithm"

  // 用例 → output 断言。tol 在 v1 一律丢弃（契约 §3.1.1）。
  const asserts = []
  ;(problem.cases || []).forEach(function (item, index) {
    if (item.tol !== null && item.tol !== undefined) {
      warnings.push("用例 " + (index + 1) + " 带 tol 参数，v1 按精确比对处理")
    }
    asserts.push({
      type: "output",
      input: item.input === undefined || item.input === null ? "" : String(item.input),
      expect: item.expect === undefined || item.expect === null ? "" : String(item.expect),
      cmp: "exact",
    })
  })

  // 契约 §3.1：「unit 为 null 当且仅当解析不出 id」；契约 §3.4：「任何单个文件解析失败只进 warnings，不中断整体加载」。
  // 所以 createUnit 抛错（id 非法等）时，**这一份题被跳过**，而不是让整个 app 起不来。
  let unit = null
  try {
    unit = createUnit({
      id: normalizeUnitId(problem.id, track),
      track: track,
      kind: "problem",
      title: problem.title || problem.id,
      statement: problem.statement || "",
      entry: {
        demo: "",
        reference: "",
        graded: "user",
        script: "",
        language: "cpp",
      },
      asserts: asserts,
      topics: problem.topics || [],
      difficulty: difficultyOf(problem.difficulty),
      source: normalizeSource(problem.source),
      status: statusOf(problem),
      progress: progressOf(problem.status),
      reviewed: false,
      reviewed_at: "",
      subtracks: problem.tracks || [],
      path: options.relPath === undefined || options.relPath === null ? "" : String(options.relPath),
    })
  } catch (err) {
    warnings.push("算法题 " + problem.id + " 解析失败：" + (err && err.message ? err.message : String(err)))
    return { unit: null, warnings: warnings }
  }

  return { unit: unit, warnings: warnings }
}

module.exports = {
  statusOf: statusOf,
  parseUnitMarkdown: parseUnitMarkdown,
}
EOF
```

```bash
const path = require("node:path")
```

```bash
/**
 * 从 README 全文里抠出第一行的 `# 标题`；抠不到就用 stem 兜底。
 *
 * @param {string} readme
 * @param {string} stem
 * @returns {string}
 */
function titleOf(readme, stem) {
  const lines = String(readme || "").split(/\r?\n/)
  for (let i = 0; i < lines.length; i += 1) {
    const m = /^#\s+(.*)$/.exec(lines[i].trim())
    if (m) {
      const title = m[1].trim()
      if (title !== "") return title
    }
  }
  return stem
}

/**
 * 读一张卡目录，产出 Unit。
 *
 * @param {string} dirAbs 卡目录的绝对路径
 * @param {object} opts { learningRoot?: string, track: "database"|"ai", unitsDir?: string|null }
 * @returns {{unit: object|null, warnings: string[]}}
 */
function parseCardDirectory(dirAbs, opts) {
  const options = opts || {}
  const track = options.track === "ai" ? "ai" : "database"
  const warnings = []
  const stem = path.basename(String(dirAbs || "").trim())
  // 前缀从 tracks.js 取（G41），不在这里手抄一份 "db:" / "ai:"。
  const t = trackOf(track)
  const prefix = t ? t.idPrefix + ":" : ""

  const readmePath = path.join(dirAbs, "README.md")
  if (!fs.existsSync(readmePath)) {
    warnings.push("卡目录没有 README.md，跳过：" + stem)
    return { unit: null, warnings: warnings }
  }
  const readme = fs.readFileSync(readmePath, "utf8")

  const notesPath = path.join(dirAbs, "NOTES.md")
  const notes = fs.existsSync(notesPath) ? fs.readFileSync(notesPath, "utf8") : ""

  // 两段都读不到内容 → 这张卡没法讲，跳过
  if (readme.trim() === "" && notes.trim() === "") {
    warnings.push("卡目录既没有 README 内容也没有 NOTES 内容，跳过：" + stem)
    return { unit: null, warnings: warnings }
  }

  const parts = []
  if (readme.trim() !== "") parts.push(readme)
  if (notes.trim() !== "") parts.push(notes)
  const statement = parts.join(SECTION_BREAK)

  const demoPath = path.join(dirAbs, "main.py")
  const hasDemo = fs.existsSync(demoPath)

  // 断言脚本的期望落点。恒填——「脚本该放哪」是卡片的属性；
  // 「脚本在不在」由 P2 的判题器判断（缺失 → compile_error）。
  const scriptRel = "units/" + track + "/" + stem + ".assert.py"
  if (options.unitsDir) {
    const scriptAbs = path.join(options.unitsDir, track, stem + ".assert.py")
    if (!fs.existsSync(scriptAbs)) {
      warnings.push("断言脚本还没写：" + scriptRel + "（P2 判题时报 compile_error）")
    }

    id: prefix + stem,
    track: track,
    kind: "card",
    title: titleOf(readme, stem),
    statement: statement,
    entry: {
      demo: hasDemo ? "scaffold/grokking/" + stem + "/main.py" : "",
      reference: "",
      graded: "script",
      script: scriptRel,
      language: "python",
    },
    asserts: [],
    topics: [],
    difficulty: "medium",
    source: "grokking",
    status: hasDemo ? "verified" : "draft",
    progress: "todo",
    reviewed: false,
    reviewed_at: "",
    subtracks: [],
    path: "scaffold/grokking/" + stem + "/",
  })

  return { unit: unit, warnings: warnings }
}

  titleOf: titleOf,
  parseCardDirectory: parseCardDirectory,
```

```bash
/**
 * 取 frontmatter 里的字符串值。
 * @param {object} fm
 * @param {string} key
 * @returns {string}
 */
function str(fm, key) {
  const value = fm[key]
  return value === undefined || value === null ? "" : String(value).trim()
}

/**
 * 解析一份单元文件，产出 Unit。
 *
 * @param {string} text 文件全文
 * @param {object} opts { path: string } —— 相对 units/ 根的路径，如 "database/mvcc.md"（必填，G42）
 * @returns {{unit: object|null, warnings: string[]}}
 */
function parseUnitFile(text, opts) {
  const options = opts || {}
  const warnings = []
  const relPath = String(options.path === undefined || options.path === null ? "" : options.path)

  // G42：opts.path 必填（契约 §3.3）。缺了就只给一条警告，不产出单元——
  // 否则 path 会退化成 "units/"：非空，却指向目录，下游永远找不到文件。
  if (relPath.trim() === "") {
    warnings.push("单元文件缺少 path，无法产出单元")
    return { unit: null, warnings: warnings }
  }

  const src = String(text || "").replace(/\r\n/g, "\n")
  let body = src
  let fm = {}
  const match = /^---\n([\s\S]*?)\n---\n?/.exec(src)
  if (match) {
    fm = parseFrontmatter(match[1])
    body = src.slice(match[0].length)
  } else {
    warnings.push("缺少 frontmatter")
  }

  const id = str(fm, "id")
  if (!ID_PATTERN.test(id)) {
    warnings.push("单元 id 非法：" + id)
    return { unit: null, warnings: warnings }
  }

  const track = str(fm, "track")
  const t = trackOf(track)
  if (!t || t.idPrefix !== id.split(":")[0]) {
    warnings.push("单元 id 前缀与 track 不一致：" + id + " / " + track)
    return { unit: null, warnings: warnings }
  }

  const kind = str(fm, "kind")
  if (KINDS.indexOf(kind) === -1) {
    warnings.push("单元 " + id + " 的 kind 非法：" + kind)
    return { unit: null, warnings: warnings }
  }

  const graded = str(fm, "graded")
  if (GRADED.indexOf(graded) === -1) {
    warnings.push("单元 " + id + " 的 graded 非法：" + graded)
    return { unit: null, warnings: warnings }
  }

  const status = str(fm, "status")
  if (STATUSES.indexOf(status) === -1) {
    warnings.push("单元 " + id + " 的 status 非法：" + status)
    return { unit: null, warnings: warnings }
  }

  const script = str(fm, "script")
  if (graded === "script" && script === "") {
    warnings.push("单元 " + id + " 声明 graded: script 但没写 script 路径")
    return { unit: null, warnings: warnings }
  }

  // 目录与 track 必须对得上（units/<track>/）
  const dirName = path.basename(path.dirname(relPath))
  if (dirName !== "" && dirName !== "." && TRACK_DIR[track] !== dirName) {
    warnings.push("单元 " + id + " 放在 units/" + dirName + "/ 下，与 track 不符")
    return { unit: null, warnings: warnings }
  }

  const sections = splitSections(body)
  // DB/AI 卡的小节叫「讲解」，算法题的小节叫「题面」——两个都认。
  const statement = String(sections["讲解"] || sections["题面"] || "").trim()
  const related = String(sections["关联"] || "").trim()

  // 算法轨单元的「测试用例」表格转成 output 断言（与 loader A 同构）
  const asserts = []
  parseCases(sections["测试用例"] || "").forEach(function (item, index) {
    // tol 在 v1 一律丢弃并留一句中文警告（与 loader A 同文案，契约 §3.1.1）。
    if (item.tol !== null && item.tol !== undefined) {
      warnings.push("用例 " + (index + 1) + " 带 tol 参数，v1 按精确比对处理")
    }
    asserts.push({
      type: "output",
      input: item.input === undefined || item.input === null ? "" : String(item.input),
      expect: item.expect === undefined || item.expect === null ? "" : String(item.expect),
      cmp: "exact",
    })
  })

  const difficulty = str(fm, "difficulty")
  const source = str(fm, "source")
  const progress = str(fm, "progress")

  let unit = null
  try {
    unit = createUnit({
      id: id,
      track: track,
      kind: kind,
      title: str(fm, "title"),
      statement: statement,
      entry: {
        demo: str(fm, "demo"),
        reference: str(fm, "reference"),
        graded: graded,
        script: graded === "script" ? script : "",
        // 算法轨的入口语言恒为 cpp；别的轨照 frontmatter 写。
        language: track === "algorithm" ? "cpp" : str(fm, "language"),
      },
      asserts: asserts,
      topics: fm.topics || [],
      difficulty: DIFFICULTIES.indexOf(difficulty) !== -1 ? difficulty : "medium",
      source: SOURCES.indexOf(source) !== -1 ? source : "generated",
      status: status,
      progress: PROGRESSES.indexOf(progress) !== -1 ? progress : "todo",
      reviewed: toBool(fm.reviewed),
      reviewed_at: str(fm, "reviewed_at"),
      subtracks: fm.subtracks || [],
      // 契约 §1.2：Unit.path 一律相对**内容根**（learning 仓根）。
      // loader A = bank/…、loader B = scaffold/grokking/…、loader C = units/<track>/<stem>.md —— 三者口径一致。
      path: "units/" + relPath,
      related: related,
    })
  } catch (err) {
    warnings.push(String(err && err.message ? err.message : err))
    return { unit: null, warnings: warnings }
  }

  return { unit: unit, warnings: warnings }
}

  TRACK_DIR: TRACK_DIR,
```

```bash
"use strict"
/**
 * index.js —— 三个 loader 的统一入口 + 单元索引缓存
 *
 * 契约 §3.4 冻结了行为：
 *   - 去重：同一 id 出现多次时先到先得，顺序是 loader C → loader A → loader B
 *   - 顺序：units 按 id 字符串升序
 *   - 不抛错：任何单个文件解析失败只进 warnings
 *
 * 为什么要这个顺序：loader C 承载的是 P4/P5 新产出的单元文件，
 * 它的 status 是执行器校验过的裁决，必须压过 loader B 的「教材即 verified」默认值。
 */

const fs = require("node:fs")
const path = require("node:path")

const loaderA = require("./loader-a")
const loaderB = require("./loader-b")
const loaderC = require("./loader-c")
const { TRACKS, GROKKING_DB_CARDS, GROKKING_AI_CARDS } = require("../track/tracks")

// 契约 §13：三条顶层轨的**唯一定义处**是 core/track/tracks.js。这里只引用，不另开一份。

/**
 * 列出目录下的 .md 文件名（不递归），按名字排序。
 * @param {string} dir
 * @returns {string[]}
 */
function listMarkdown(dir) {
  let names = []
  try {
    names = fs.readdirSync(dir)
  } catch (e) {
    return []
  }
  return names
    .filter(function (name) {
      return name.endsWith(".md")
    })
    .sort()
}

/**
 * 读一个文本文件；读不到返回 null。
 * @param {string} abs
 * @returns {string|null}
 */
function readText(abs) {
  try {
    return fs.readFileSync(abs, "utf8")
  } catch (e) {
    return null
  }
}

/**
 * 扫 loader C 的单元文件。
 * @param {string|null} unitsDir learning/units 的绝对路径
 * @param {string[]} warnings 收集警告
 * @returns {object[]}
 */
function loadFromUnitsDir(unitsDir, warnings) {
  const units = []
  if (!unitsDir) return units
  TRACKS.forEach(function (track) {
    // track 是**对象**，轨道代码取 track.code（不是 track 本身）。
    const dir = path.join(unitsDir, track.code)
    listMarkdown(dir).forEach(function (name) {
      const text = readText(path.join(dir, name))
      if (text === null) {
        warnings.push("读不了单元文件：" + name)
        return
      }
      const res = loaderC.parseUnitFile(text, { path: track.code + "/" + name })
      ;(res.warnings || []).forEach(function (w) {
        warnings.push(w)
      })
      if (res.unit) units.push(res.unit)
    })
  })
  return units
}

/**
 * 扫 loader A 的题库目录（bank/<子轨>/*.md）。
 * @param {string|null} bankRoot 题库根的绝对路径
 * @param {string[]} warnings
 * @returns {object[]}
 */
function loadFromBank(bankRoot, warnings) {
  const units = []
  if (!bankRoot) return units
  let subs = []
  try {
    subs = fs
      .readdirSync(bankRoot, { withFileTypes: true })
      .filter(function (item) {
        return item.isDirectory()
      })
      .map(function (item) {
        return item.name
      })
      .sort()
  } catch (e) {
    warnings.push("题库目录读不到：" + bankRoot)
    return units
  }
  subs.forEach(function (sub) {
    const dir = path.join(bankRoot, sub)
    listMarkdown(dir).forEach(function (name) {
      const text = readText(path.join(dir, name))
      if (text === null) {
        warnings.push("读不了题库文件：" + name)
        return
      }
      const res = loaderA.parseUnitMarkdown(text, { track: "algorithm", relPath: "bank/" + sub + "/" + name })
      ;(res.warnings || []).forEach(function (w) {
        warnings.push(w)
      })
      if (res.unit) units.push(res.unit)
    })
  })
  return units
}

/**
 * 扫 loader B 的 grokking 卡。
 *
 * 只扫契约 §14.1 冻结的 29 张卡（db 16 + ai 13）——
 * grokking 目录下还有 59 张纯算法 / C++ 并发卡，不属本轮三轨范围。
 *
 * @param {string|null} grokkingDir learning/scaffold/grokking 的绝对路径
 * @param {string|null} unitsDir learning/units 的绝对路径
 * @param {string[]} warnings
 * @returns {object[]}
 */
function loadFromCards(grokkingDir, unitsDir, warnings) {
  const units = []
  if (!grokkingDir) return units
  const cards = []
  GROKKING_DB_CARDS.forEach(function (stem) {
    cards.push({ stem: stem, track: "database" })
  })
  GROKKING_AI_CARDS.forEach(function (stem) {
    cards.push({ stem: stem, track: "ai" })
  })
  cards.forEach(function (card) {
    const dir = path.join(grokkingDir, card.stem)
    if (!fs.existsSync(dir)) {
      warnings.push("卡目录不存在：" + card.stem)
      return
    }
    const res = loaderB.parseCardDirectory(dir, {
      learningRoot: path.dirname(path.dirname(grokkingDir)),
      track: card.track,
      unitsDir: unitsDir,
    })
    ;(res.warnings || []).forEach(function (w) {
      warnings.push(w)
    })
    if (res.unit) units.push(res.unit)
  })
  return units
}

/**
 * 加载全部单元。
 *
 * @param {object} opts { learningRoot, localBank, unitsDir, grokkingDir }
 * @returns {{units: object[], warnings: string[], byId: object}}
 */
function loadUnits(opts) {
  const options = opts || {}
  const warnings = []

  // 去重顺序：C → A → B
  const collected = []
    .concat(loadFromUnitsDir(options.unitsDir, warnings))
    .concat(loadFromBank(options.learningRoot ? path.join(options.learningRoot, "bank") : null, warnings))
    .concat(loadFromBank(options.localBank, warnings))
    .concat(loadFromCards(options.grokkingDir, options.unitsDir, warnings))

  const byId = {}
  const units = []
  collected.forEach(function (unit) {
    if (Object.prototype.hasOwnProperty.call(byId, unit.id)) {
      warnings.push("单元 id 重复：" + unit.id + "（保留先出现的）")
      return
    }
    byId[unit.id] = unit
    units.push(unit)
  })

  units.sort(function (a, b) {
    return a.id < b.id ? -1 : a.id > b.id ? 1 : 0
  })

  return { units: units, warnings: warnings, byId: byId }
}

/**
 * 把 loadUnits 的结果物化进 units 表（整体覆盖语义）。
 *
 * 依赖 store.replaceUnits / store.listUnits（server/db.js 提供）。
 * 不抛错：任何加载失败只进 warnings。
 *
 * @param {object} store
 * @param {object} opts 同 loadUnits
 * @returns {{count: number, warnings: string[]}}
 */
function refreshUnitIndex(store, opts, preloaded) {
  // 契约 §3.4：第三个可选参数 preloaded —— 调用方已经有 loadUnits 的结果时直接传进来，
  // 本函数**不再扫盘**（启动时扫一次就够，别把 700+ 份 md 解析两遍）。
  const res = preloaded && Array.isArray(preloaded.units)
    ? preloaded
    : loadUnits(opts)
  const now = new Date().toISOString()
  const rows = res.units.map(function (unit) {
    return {
      id: unit.id,
      track: unit.track,
      kind: unit.kind,
      title: unit.title,
      path: unit.path,
      status: unit.status,
      topics: JSON.stringify(unit.topics || []),
      difficulty: unit.difficulty,
      subtracks: JSON.stringify(unit.subtracks || []),
      updated: now,
    }
  })
  store.replaceUnits(rows)
  return { count: rows.length, warnings: res.warnings }
}

module.exports = {
  TRACKS: TRACKS,
  listMarkdown: listMarkdown,
  loadUnits: loadUnits,
  refreshUnitIndex: refreshUnitIndex,
}
```

- [ ] **Step 4: 在 `server/db.js` 里加 `migrate`**

在 `openDb` 函数**之前**插入下面这段，并在文件顶部的常量区加上 `ADD_COLUMNS`：

```js
/**
 * P0 要加的列。纯加法：只加，不改既有列，不动既有数据。
 * NOT NULL 的列必须带 DEFAULT，否则老库 ALTER 会失败。
 */
const ADD_COLUMNS = [
  { table: "attempts", column: "unit_id", type: "TEXT" },
  { table: "attempts", column: "track", type: "TEXT NOT NULL DEFAULT 'algorithm'" },
  { table: "attempts", column: "assert_total", type: "INTEGER NOT NULL DEFAULT 0" },
  { table: "attempts", column: "assert_passed", type: "INTEGER NOT NULL DEFAULT 0" },
  { table: "review_cards", column: "unit_id", type: "TEXT" },
  { table: "review_cards", column: "units_since", type: "INTEGER NOT NULL DEFAULT 0" },
  { table: "topics", column: "track", type: "TEXT NOT NULL DEFAULT 'algorithm'" },
]

/**
 * 迁移：给老库补上缺的列。
 *
 * 幂等：同一列第二次跑会被跳过，所以 openDb 对同一个文件连开两次结果一致。
 * **P0 阶段一条数据都不改**——数据迁移留给 P1（attempts/review_cards 的 unit_id）
 * 与 P3（topics 的复合键化）。
 *
 * @param {import("node:sqlite").DatabaseSync} db
 * @returns {{applied: string[], warnings: string[]}}
 */
function migrate(db) {
  const applied = []
  const warnings = []
  ADD_COLUMNS.forEach(function (item) {
    let info = []
    try {
      info = db.prepare("PRAGMA table_info(" + item.table + ")").all()
    } catch (err) {
      warnings.push("读不到表结构：" + item.table)
      return
    }
    const exists = info.some(function (row) {
      return String(row.name) === item.column
    })
    if (exists) return
    const stmt = "ALTER TABLE " + item.table + " ADD COLUMN " + item.column + " " + item.type
    try {
      db.exec(stmt)
      applied.push(stmt)
    } catch (err) {
      warnings.push("加列失败：" + item.table + "." + item.column + "：" + String(err && err.message ? err.message : err))
    }
  })
  return { applied: applied, warnings: warnings }
}
```

```js
function openDb(filePath) {
  const target = filePath || ":memory:"
  if (target !== ":memory:") {
    fs.mkdirSync(path.dirname(target), { recursive: true })
  }
  const db = new DatabaseSync(target)
  SCHEMA.forEach(function (stmt) {
    db.exec(stmt)
  })
  // 老库补列。必须在建表之后跑：新库由 SCHEMA 直接建成带新列的形状，
  // 老库的表已经存在（CREATE TABLE IF NOT EXISTS 不会改它），只能靠 ALTER 补齐。
  migrate(db)
  return db
}
```

```js
/**
 * 解析一个 JSON 数组字符串；不是合法 JSON 就当空数组。
 * @param {*} text
 * @returns {string[]}
 */
function parseJsonList(text) {
  const raw = String(text === undefined || text === null ? "" : text)
  if (raw === "") return []
  try {
    const value = JSON.parse(raw)
    if (!Array.isArray(value)) return []
    return value.map(function (item) {
      return String(item)
    })
  } catch (e) {
    return []
  }
}
```

```bash
"use strict"
/**
 * api-units.js —— 单元查询接口（P0 新增）
 *
 * 契约 §8.1：
 *   GET /api/units     → { total, items: [...] }
 *   GET /api/units/:id → { unit, statement_html }
 *
 * 列表走 units 表（物化视图，不每次扫盘）；详情走启动时载入的单元快照
 * （表里只有摘要字段，全字段的 Unit 与 statement 得从 loader 拿）。
 */

const { toHtml } = require("../core/unit/markdown.js")

/**
 * 组装路由表。
 * @param {{store: object, unitSnapshot: object, sendJson: Function}} ctx
 * @returns {Array<{method: string, pattern: string, handler: Function}>}
 */
function buildUnitRoutes(ctx) {
  const routes = []
  const store = ctx.store
  const snapshot = ctx.unitSnapshot || { units: [], warnings: [], byId: {} }

  routes.push({
    method: "GET",
    pattern: "/api/units",
    handler: async function (req, res, params, query) {
      const filter = {}
      const track = query.get("track")
      if (track) filter.track = track
      const kind = query.get("kind")
      if (kind) filter.kind = kind
      const status = query.get("status")
      if (status) filter.status = status

      let rows = store.listUnits(filter)

      // q 是标题的模糊搜索。放在 SQL 之外做：表里没有别名字段可搜，
      // 而 units 的规模（几千行）内存过滤完全够快。
      const q = query.get("q")
      if (q) {
        const needle = String(q).toLowerCase()
        rows = rows.filter(function (row) {
          return String(row.title).toLowerCase().indexOf(needle) !== -1
        })
      }

      const total = rows.length

      // limit 非正整数 → 按不限制处理（返回全部匹配项）
      const limitRaw = query.get("limit")
      const limit = limitRaw === null ? NaN : Number(limitRaw)
      if (Number.isFinite(limit) && limit > 0) {
        rows = rows.slice(0, limit)
      }

      ctx.sendJson(res, 200, { total: total, items: rows })
    },
  })

  routes.push({
    method: "GET",
    pattern: "/api/units/:id",
    handler: async function (req, res, params) {
      const id = String(params.id || "")
      const unit = snapshot.byId[id]
      if (!unit) {
        ctx.sendJson(res, 404, { error: "单元不存在", id: id })
        return
      }
      ctx.sendJson(res, 200, {
        unit: unit,
        statement_html: toHtml(unit.statement || ""),
      })
    },
  })

  return routes
}

module.exports = { buildUnitRoutes: buildUnitRoutes }
```

**Interfaces:**

- Produces: 4 个断言脚本，形状冻结（P4 生成的脚本照此模板）：
  - 全部断言通过 → `exit 0`；任一条不通过 → 打印失败原因并 `exit 1`
  - demo 跑在**临时沙箱**里，`.db` 文件不落 learning 仓的工作树

> **为什么断言脚本用 `subprocess` 而不是 `import`**：`mvcc` / `transaction` / `index_design` / `sql_fundamentals`

> 所以脚本把 demo 复制进一个临时目录再跑——每次都是干净环境，`.db` 也只落在临时目录里。

```bash
function resolveLearningRoot() {
  const fromEnv = process.env.STUDY_HUB_LEARNING_ROOT
  if (fromEnv && fs.existsSync(fromEnv)) return path.resolve(fromEnv)
  const fallback = path.resolve(ROOT, "..", "..", "..", "learning")
  if (fs.existsSync(fallback)) return fallback
  return null
}

/**
 * 找一个能用的 Python 命令。
 * @returns {string|null}
 */
function resolvePython() {
  const candidates = ["python3", "python"]
  for (const cmd of candidates) {
    const probe = spawnSync(cmd, ["--version"], { encoding: "utf8" })
    if (!probe.error && probe.status === 0) return cmd
  }
  return null
}
```
